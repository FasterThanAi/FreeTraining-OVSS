\section{Introduction}
\begin{frame}{\textsc{1. Introduction}}
\small
\textbf{Open-vocabulary semantic segmentation of aerial imagery.} The input is an
image together with a list of class names given as plain words; the output is a
class label for every pixel. The setting is \best{training-free}: no network
weights are learned, and the class list may change at inference without
retraining.

\vspace{0.4em}
\textbf{Baseline.} SegEarth-OV3~\cite{segearthov3} applies SAM~3~\cite{sam3} to
remote sensing. For each class prompt, SAM~3 emits a presence score, a dense
semantic map and a set of instance masks; SegEarth-OV3 fuses these and applies a
single confidence threshold $\tau$, tuned per dataset. Pixels below $\tau$ are
assigned to a catch-all class. \ok{Reproduced on our hardware at 47.38 mIoU
against a published 47.4.}

\vspace{0.6em}
\begin{block}{Contribution}
A single confidence threshold is the wrong \emph{shape} for a multi-class
pipeline. We fit \best{one threshold per class}, and \best{one scale per class
applied before the \texttt{argmax}}, using the same supervision budget the
baseline already spends tuning its single $\tau$. \best{No network weights are
trained.}
\end{block}
\end{frame}

% ===================================================================== 2
\section{Motivation \& Applications}
\begin{frame}{\textsc{2. Motivation --- the baseline discards land cover}}
\small
Measured on LoveDA validation at SegEarth-OV3's own published $\tau = 0.5$:
\begin{itemize}
  \item \best{29.68\%} of all real-class pixels are assigned to the catch-all
        class \bg{} --- \best{323 million pixels}.
  \item Discard exceeds real-class confusion by \best{3:1}. The dominant error
        mode is abstention rather than misclassification.
  \item \texttt{water}: precision \best{89.5\%}, recall \best{54.7\%}. The class
        is correct nine times in ten when predicted, yet recovers barely half of
        the water present.
\end{itemize}
\vspace{0.5em}
\begin{block}{Cost of global remedies}
\begin{tabular}{@{}p{4.6cm}l@{}}
Lower $\tau$ to 0.1 & \bad{$-5.54$ mIoU}; 1.73 incorrect pixels per correct one \\
Remove presence gating & \bad{$-11.97$ mIoU} \\
No intervention & 29.68\% of land cover discarded \\
\end{tabular}
\end{block}
\vspace{0.4em}
A correction that operates uniformly across classes therefore cannot recover the
residual. \best{The correction must be per-class.}
\end{frame}

\begin{frame}{\textsc{2. Motivation --- the discard, visualised}}
\begin{center}
\includegraphics[width=0.97\textwidth]{2524.png}
\end{center}
\vspace{0.4em}
\footnotesize A LoveDA validation tile. Left to right: \textbf{input image},

\begin{frame}{\textsc{2. Application domains}}
\small
The setting is intended for deployments in which the class list is not fixed in
advance and labelled data for the target region does not exist.
\vspace{0.5em}
\begin{itemize}
  \item \textbf{Disaster response.} Flood or damage extent from UAV imagery
        acquired within hours of an event, where no annotated data exists for the
        affected district and there is no time to train a model.
  \vspace{0.3em}
  \item \textbf{Infrastructure monitoring.} Classes such as
        \texttt{solar panel} or \texttt{transmission line} appear in no
        land-cover benchmark, yet can be supplied to the pipeline as words.
  \vspace{0.3em}
  \item \textbf{Settlement mapping.} Vocabularies such as \texttt{tin roof},
        \texttt{unpaved road} and \texttt{water tank} change between surveys and
        between regions.
\end{itemize}
\vspace{0.6em}
\begin{block}{Deployment cost}
The proposed correction requires \best{$\sim$\calibTiles{} labelled tiles} and
\best{no gradient steps}. Adapting the same pipeline by fine-tuning would require
labelled data in the thousands of tiles, a training run, and would forfeit the
training-free property.
\end{block}
\end{frame}

\section{Problem Statement}
\begin{frame}{\textsc{3. Problem statement --- formulation}}
\small
Let $\mathcal{C}=\{1,\dots,N\}$ denote the vocabulary and $s_c(p)\in[0,1]$ the
score the pipeline assigns to class $c$ at pixel $p$. Let $b$ denote the
catch-all class.

\vspace{0.4em}
\textbf{Baseline decision rule}, with a single global threshold $\tau$:
\[
  \hat{y}(p)=
  \begin{cases}
    \arg\max_{c} s_c(p), & \text{if } \max_c s_c(p) \ge \tau\\[2pt]
    b, & \text{otherwise}
  \end{cases}
\]

\textbf{Proposed rule}, with parameters fitted on a calibration set of
$\sim$\calibTiles{} labelled tiles:
\[
  \hat{y}(p)=
  \begin{cases}
    c^\star, & \text{if } s_{c^\star}(p) \ge \boldsymbol{\tau_{c^\star}}\\
    b, & \text{otherwise}
  \end{cases}
  \qquad
  c^\star = \arg\max_{c}\ \boldsymbol{w_c}\, s_c(p)
\]

\vspace{0.3em}
\begin{block}{Objective}
$(\boldsymbol{w},\boldsymbol{\tau}) = \arg\max\ \operatorname{mIoU}_{\text{real}}$,
optimised by coordinate ascent. \best{No network weights are modified.} The
threshold is applied to the \emph{raw} score, so $\boldsymbol{w}$ influences only
the \texttt{argmax}.
\end{block}
\end{frame}

\begin{frame}{\textsc{3. Completeness of the per-class threshold family}}
\small
\textbf{Proposition.} For a fixed \texttt{argmax}, any strictly increasing
per-class recalibration $g_c$ composed with a global threshold $\tau$ is
equivalent to a per-class threshold vector:
\[
  g_c(s)\ \ge\ \tau \quad\Longleftrightarrow\quad s\ \ge\ g_c^{-1}(\tau)\;=\;\tau_c .
\]
\begin{itemize}
  \item \best{Per-class thresholds therefore exhaust every correction reachable
        by rescaling scores after the \texttt{argmax}}; no member of that family
        can outperform them.
  \item Scaling applied \emph{before} the \texttt{argmax} alters which class
        wins, and so constitutes a \best{strictly larger family}.
\end{itemize}
\vspace{0.4em}
\begin{block}{Empirical consequence}
On LoveDA, \best{6.0\%} of catch-all assignments are pixels for which the
catch-all class won the \texttt{argmax} at $s \ge \tau$. A threshold cannot alter
an \texttt{argmax}, so this residual is unreachable by the first parameter set
and motivates the second.
\end{block}
\end{frame}

\section{Research Gaps}
\begin{frame}{\textsc{5. Research gaps}}
\small
\begin{block}{Gap 1 --- the decision rule is unexamined}
Every method in this family terminates in an \texttt{argmax} followed by a single
global threshold, tuned per dataset. Research effort is directed at improving the
\emph{scores}; whether the \emph{decision rule} applied to them has the
appropriate form has not been studied.
\end{block}
\begin{block}{Gap 2 --- the discarded residual is not quantified}
Published work reports mIoU. It does not report what fraction of real land cover
is assigned to the catch-all class, nor whether that fraction is recoverable.
\end{block}
\begin{block}{Gap 3 --- catch-all classes distort the reported metric}
mIoU is an \emph{unweighted} mean over classes, so a catch-all class contributes
$1/N$ of the score irrespective of its semantic content. A method may therefore
appear to improve or degrade segmentation while land-cover quality is unchanged.
\end{block}
\end{frame}

\section{Objectives}
\begin{frame}{\textsc{6. Objectives}}
\small
\begin{block}{Completed}
\begin{enumerate}
  \item \textbf{Reproduce} SegEarth-OV3 on local hardware, so that every
        subsequent measurement is made against a baseline we control.
        \hfill \ok{47.38 vs published 47.4}
  \item \textbf{Quantify} the residual the baseline discards, and determine
        whether any global adjustment recovers it.
        \hfill \ok{29.68\%; no global adjustment does}
  \item \textbf{Design} a correction that trains no weights and consumes only the
        supervision budget the baseline already spends.
        \hfill \ok{two parameter sets}
\end{enumerate}
\end{block}
\begin{block}{In progress}
\begin{enumerate}
  \setcounter{enumi}{3}
  \item \textbf{Extend} the correction beyond a single vector per dataset:
        the fitted parameters are currently constant across tiles, and a
        scene-adaptive formulation remains to be investigated.
  \item \textbf{Investigate} a per-class fusion of SAM~3's semantic and instance
        heads, which are presently combined by a fixed maximum.
  \item \textbf{Broaden} the evaluation to further high-resolution benchmarks and
        establish the conditions under which the correction transfers.
\end{enumerate}
\end{block}
\end{frame}

  \item Per-class Otsu and eleven further label-free rules
\end{itemize}

\begin{column}{0.48\textwidth}
\textbf{Protocol}
\begin{itemize}\footnotesize
  \item Five-fold cross-validation. Calibration and evaluation tiles are
        \best{always disjoint}.
  \item The baseline is recomputed on the \emph{same} held-out tiles, so both
        rules are scored on identical pixels.
  \item Both metrics are reported: full mIoU and mIoU excluding the catch-all
        class.
  \item Results are confirmed by the official evaluator, not by arithmetic on a
        cached statistic.
\end{itemize}
\vspace{0.5em}
\begin{block}{\small Validation gate}
\footnotesize Every instrumented run must still reproduce \best{47.37 mIoU} and
\best{29.68\%} discard. A deviation indicates that the instrumentation altered
model behaviour.
\end{block}
\end{column}

\section{Application Study}
\begin{frame}{\textsc{8. Application study --- an unseen region and vocabulary}}
\small
The correction was applied to \indTiles{} tiles cut from \indScenes{} openly
licensed UAV scenes over India at \indGSD{} ground sample distance --- a region
present in none of the three benchmarks --- under two class lists supplied as
words at inference.
\vspace{0.4em}
\begin{itemize}
  \item \ok{The segmentation is stable under renaming.} A flooded building is
        partitioned almost identically by both vocabularies:
        \indFloodA{}\% \texttt{water} against \indFloodB{}\% \texttt{flood water},
        and \indBldgA{}\% \texttt{concrete roof} against \indBldgB{}\%
        \texttt{building}. The regions are recovered consistently; only the
        labels follow the supplied words.
  \item \bad{A class absent from the vocabulary is not recovered.}
        \indVegA{}\% of a photovoltaic-array tile is assigned to a single class
        until \texttt{solar panel} is supplied, which then accounts for
        \indSolarB{}\% --- below the visible extent of the panels.
\end{itemize}
\vspace{0.4em}
\footnotesize \bad{No ground truth exists for this imagery.} The quantities above
are per-class pixel shares and are not accuracy measurements.
\end{frame}
